# Sirius Implementation Plan

> **For Claude:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task.

**Goal:** Build Sirius — a Meridian-derived DLMM sidecar that *alerts* on 15m-Supertrend entries (operator deploys by hand), *auto-closes* positions on a TA exit confluence, and *alerts* (no close) on dumps.

**Architecture:** Polaris-style clone of Meridian at `D:\aiproject\sirius` with its own repo/wallet/config. Entry loop is read-only alerting; exit loop is the only on-chain write (`close` + post-close claim/swap). `deploy_position` is removed entirely. New pure functions (MACD, `ta_exit` preset, inverted dump detector) are TDD'd against known candle data; loop wiring reuses existing Meridian infra.

**Tech Stack:** Node ESM, `@meteora-ag/dlmm`, Agent Meridian chart-indicators relay (298 candles, client-side MACD), Telegram Bot API. Tests are standalone `node test-*.js` scripts (force-added; `test-*.js` is gitignored per convention).

```js
// Client-side MACD (12/26/9) computed from a close-price series.
function ema(values, period) {
  const k = 2 / (period + 1);
  const out = new Array(values.length).fill(null);
  let prev = null;
  for (let i = 0; i < values.length; i++) {
    const v = Number(values[i]);
    if (!Number.isFinite(v)) { out[i] = prev; continue; }
    prev = prev == null ? v : v * k + prev * (1 - k);
    out[i] = prev;
  }
  return out;
}

export function computeMacd(closes, { fast = 12, slow = 26, signal = 9 } = {}) {
  const emaFast = ema(closes, fast);
  const emaSlow = ema(closes, slow);
  const macdLine = closes.map((_, i) =>
    emaFast[i] != null && emaSlow[i] != null ? emaFast[i] - emaSlow[i] : null);
  const macdSignal = ema(macdLine.map((v) => (v == null ? 0 : v)), signal);
  const hist = macdLine.map((v, i) =>
    v != null && macdSignal[i] != null ? v - macdSignal[i] : null);
  const prevHist = hist.at(-2);
  const lastHist = hist.at(-1);
  const firstGreen =
    prevHist != null && lastHist != null && prevHist <= 0 && lastHist > 0;
  return { macdLine, macdSignal, hist, prevHist, lastHist, firstGreen };
}
```

**Step 3: Implement** (drops are negative percents; trip when drop ≤ threshold):

```js
export function detectRecentDump(candles, { maxSingleDropPct, maxWindowDropPct } = {}) {
  if (!Array.isArray(candles) || candles.length === 0) return null;
  let minSingle = 0;
  for (const c of candles) {
    const o = Number(c?.open), cl = Number(c?.close);
    if (!(o > 0) || !(cl > 0)) continue;
    const chg = (cl / o - 1) * 100;
    if (chg < minSingle) minSingle = chg;
  }
  let minWindow = 0;
  for (let i = 2; i < candles.length; i++) {
    const base = Number(candles[i - 2]?.open), end = Number(candles[i]?.close);
    if (!(base > 0) || !(end > 0)) continue;
    const chg = (end / base - 1) * 100;
    if (chg < minWindow) minWindow = chg;
  }
  const single = maxSingleDropPct != null && minSingle <= maxSingleDropPct;
  const windowed = maxWindowDropPct != null && minWindow <= maxWindowDropPct;
  return { dumped: single || windowed, minSingle5mPct: Number(minSingle.toFixed(1)), minWindowPct: Number(minWindow.toFixed(1)) };
}
```
